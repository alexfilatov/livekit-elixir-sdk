defmodule Livekit.Agents.Pipeline.EnergyVADTest do
  use ExUnit.Case, async: true

  alias Livekit.Agents.AudioFrame
  alias Livekit.Agents.Pipeline.EnergyVAD

  # Helpers
  defp silence_frame do
    # 320 bytes of zeros = 160 pcm_16 samples, all zero -> RMS = 0.0
    AudioFrame.new(<<0::320*8>>, sample_rate: 16_000, format: :pcm_16)
  end

  defp speech_frame do
    # 320 bytes of max-amplitude samples -> RMS ~1.0
    data = for _ <- 1..160, into: <<>>, do: <<32767::little-signed-16>>
    AudioFrame.new(data, sample_rate: 16_000, format: :pcm_16)
  end

  describe "new/1" do
    test "uses default threshold 0.01 when no opts given" do
      cfg = EnergyVAD.new(%{})
      assert cfg.threshold == 0.01
    end

    test "uses provided threshold" do
      cfg = EnergyVAD.new(%{threshold: 0.05})
      assert cfg.threshold == 0.05
    end
  end

  describe "classify/2" do
    test "returns :silence for a zero-amplitude frame" do
      cfg = EnergyVAD.new(%{})
      assert EnergyVAD.classify(silence_frame(), cfg) == :silence
    end

    test "returns :speech for a high-amplitude frame" do
      cfg = EnergyVAD.new(%{})
      assert EnergyVAD.classify(speech_frame(), cfg) == :speech
    end

    test "threshold controls the boundary: high threshold classifies speech as silence" do
      # threshold above RMS of a mid-amplitude frame -> silence
      mid_data = for _ <- 1..160, into: <<>>, do: <<100::little-signed-16>>
      mid_frame = AudioFrame.new(mid_data, sample_rate: 16_000, format: :pcm_16)
      high_cfg = EnergyVAD.new(%{threshold: 0.99})
      assert EnergyVAD.classify(mid_frame, high_cfg) == :silence
    end

    test "low threshold classifies mid-amplitude frame as speech" do
      mid_data = for _ <- 1..160, into: <<>>, do: <<100::little-signed-16>>
      mid_frame = AudioFrame.new(mid_data, sample_rate: 16_000, format: :pcm_16)
      low_cfg = EnergyVAD.new(%{threshold: 0.001})
      assert EnergyVAD.classify(mid_frame, low_cfg) == :speech
    end
  end
end

defmodule Livekit.Agents.Pipeline.TurnDetectorTest do
  use ExUnit.Case, async: true

  alias Livekit.Agents.AudioFrame
  alias Livekit.Agents.Pipeline.TurnDetector

  defp silence_frame, do: AudioFrame.new(<<0::320*8>>, sample_rate: 16_000, format: :pcm_16)
  defp speech_frame do
    data = for _ <- 1..160, into: <<>>, do: <<32767::little-signed-16>>
    AudioFrame.new(data, sample_rate: 16_000, format: :pcm_16)
  end

  defp start_detector(silence_ms \\ 50) do
    {:ok, pid} = TurnDetector.start_link(silence_ms: silence_ms, subscriber: self())
    pid
  end

  describe "turn_start" do
    test "sends {:turn_start, timestamp_us} on first speech frame" do
      pid = start_detector()
      frame = %{speech_frame() | timestamp_us: 12345}
      TurnDetector.push_frame(pid, {:speech, frame})
      assert_receive {:turn_start, 12345}, 200
    end

    test "does NOT send :turn_start on consecutive speech frames (already speaking)" do
      pid = start_detector()
      frame = speech_frame()
      TurnDetector.push_frame(pid, {:speech, frame})
      assert_receive {:turn_start, _}, 200
      TurnDetector.push_frame(pid, {:speech, frame})
      refute_receive {:turn_start, _}, 100
    end
  end

  describe "turn_end" do
    test "sends {:turn_end, frames} after silence window elapses" do
      pid = start_detector(50)
      frame = speech_frame()
      TurnDetector.push_frame(pid, {:speech, frame})
      assert_receive {:turn_start, _}, 200
      TurnDetector.push_frame(pid, {:silence, silence_frame()})
      # turn_end arrives after silence_ms (50ms) elapses
      assert_receive {:turn_end, frames}, 500
      assert is_list(frames)
      assert length(frames) == 1
    end

    test ":turn_end includes all accumulated speech frames" do
      pid = start_detector(50)
      f1 = speech_frame()
      f2 = speech_frame()
      TurnDetector.push_frame(pid, {:speech, f1})
      assert_receive {:turn_start, _}, 200
      TurnDetector.push_frame(pid, {:speech, f2})
      TurnDetector.push_frame(pid, {:silence, silence_frame()})
      assert_receive {:turn_end, frames}, 500
      assert length(frames) == 2
    end

    test "silence frame before any speech produces no :turn_end" do
      pid = start_detector(50)
      TurnDetector.push_frame(pid, {:silence, silence_frame()})
      refute_receive {:turn_end, _}, 200
    end
  end

  describe "timer reset on mid-silence speech" do
    test "speech during silence window cancels :turn_end and sends new :turn_start only if was speaking" do
      pid = start_detector(80)
      # Speak, then silence but immediately speak again before timer fires
      TurnDetector.push_frame(pid, {:speech, speech_frame()})
      assert_receive {:turn_start, _}, 200
      TurnDetector.push_frame(pid, {:silence, silence_frame()})
      # Immediately push speech before 80ms elapses
      TurnDetector.push_frame(pid, {:speech, speech_frame()})
      # turn_end should NOT fire yet
      refute_receive {:turn_end, _}, 40
      # After silence window, the final turn_end fires
      TurnDetector.push_frame(pid, {:silence, silence_frame()})
      assert_receive {:turn_end, _}, 500
    end
  end

  describe "reset/1" do
    test "reset cancels pending timer and clears frames" do
      pid = start_detector(200)
      TurnDetector.push_frame(pid, {:speech, speech_frame()})
      assert_receive {:turn_start, _}, 200
      TurnDetector.push_frame(pid, {:silence, silence_frame()})
      TurnDetector.reset(pid)
      # turn_end should not arrive after reset
      refute_receive {:turn_end, _}, 400
    end
  end
end

defmodule Livekit.Agents.PipelineTest do
  use ExUnit.Case, async: false  # telemetry attach is global

  alias Livekit.Agents.{AudioFrame, Pipeline}

  alias Livekit.Agents.Pipeline.Test.{MockSTT, MockLLM, MockTTS, FailingSTT}

  defp speech_data do
    for _ <- 1..160, into: <<>>, do: <<32767::little-signed-16>>
  end

  defp silence_data, do: <<0::320*8>>

  defp speech_frame, do: AudioFrame.new(speech_data(), sample_rate: 16_000, format: :pcm_16)
  defp silence_frame, do: AudioFrame.new(silence_data(), sample_rate: 16_000, format: :pcm_16)

  defp base_config(overrides \\ []) do
    %Pipeline.Config{
      stt: {MockSTT, %{}},
      llm: {MockLLM, %{}},
      tts: {MockTTS, %{}},
      silence_ms: Keyword.get(overrides, :silence_ms, 50)
    }
  end

  describe "start_link/1" do
    test "starts successfully with valid config" do
      {:ok, pid} = Pipeline.start_link(base_config())
      assert Process.alive?(pid)
      Pipeline.stop(pid)
    end

    test "returns error when stt is nil" do
      config = %Pipeline.Config{stt: nil, llm: {MockLLM, %{}}, tts: {MockTTS, %{}}}
      assert {:error, _} = Pipeline.start_link(config)
    end
  end

  describe "push_frame/2" do
    test "returns :ok immediately (non-blocking)" do
      {:ok, pid} = Pipeline.start_link(base_config())
      result = Pipeline.push_frame(pid, speech_frame())
      assert result == :ok
      Pipeline.stop(pid)
    end
  end

  describe "full STT->LLM->TTS integration turn (TEST-07)" do
    test "pushing speech frames then silence completes a full pipeline turn" do
      {:ok, pid} = Pipeline.start_link(base_config(silence_ms: 50))

      # Push speech frames to accumulate an utterance
      Enum.each(1..5, fn _ -> Pipeline.push_frame(pid, speech_frame()) end)
      # Push silence to trigger end-of-turn
      Pipeline.push_frame(pid, silence_frame())

      # Wait for async Task to complete (STT + LLM + TTS)
      Process.sleep(300)

      metrics = Pipeline.get_metrics(pid)
      assert metrics.turns_processed == 1
      assert metrics.errors == 0
      assert metrics.audio_frames_processed == 6

      Pipeline.stop(pid)
    end

    test "conversation context accumulates across turns" do
      {:ok, pid} = Pipeline.start_link(base_config(silence_ms: 50))

      # First turn
      Pipeline.push_frame(pid, speech_frame())
      Pipeline.push_frame(pid, silence_frame())
      Process.sleep(300)

      # Second turn
      Pipeline.push_frame(pid, speech_frame())
      Pipeline.push_frame(pid, silence_frame())
      Process.sleep(300)

      metrics = Pipeline.get_metrics(pid)
      assert metrics.turns_processed == 2

      Pipeline.stop(pid)
    end
  end

  describe ":telemetry events (PIPE-06)" do
    test "stt_complete, llm_first_token, tts_start events fire in order during a turn" do
      test_pid = self()
      events_received = :ets.new(:events, [:ordered_set, :public])

      handler_id = "test-pipeline-#{System.unique_integer()}"

      events = [
        [:livekit, :agents, :pipeline, :stt_complete],
        [:livekit, :agents, :pipeline, :llm_first_token],
        [:livekit, :agents, :pipeline, :tts_start]
      ]

      :telemetry.attach_many(
        handler_id,
        events,
        fn event_name, measurements, _metadata, _config ->
          :ets.insert(events_received, {System.monotonic_time(), event_name, measurements})
          send(test_pid, {:telemetry_event, List.last(event_name)})
        end,
        nil
      )

      {:ok, pid} = Pipeline.start_link(base_config(silence_ms: 50))
      Pipeline.push_frame(pid, speech_frame())
      Pipeline.push_frame(pid, silence_frame())

      assert_receive {:telemetry_event, :stt_complete}, 500
      assert_receive {:telemetry_event, :llm_first_token}, 500
      assert_receive {:telemetry_event, :tts_start}, 500

      # Verify order
      ordered = :ets.tab2list(events_received) |> Enum.sort() |> Enum.map(&elem(&1, 1))
      assert ordered == [
        [:livekit, :agents, :pipeline, :stt_complete],
        [:livekit, :agents, :pipeline, :llm_first_token],
        [:livekit, :agents, :pipeline, :tts_start]
      ]

      :telemetry.detach(handler_id)
      Pipeline.stop(pid)
    end
  end

  describe "interruption handling (PIPE-04)" do
    test "new speech while processing cancels active task and allows new turn to complete" do
      {:ok, pid} = Pipeline.start_link(base_config(silence_ms: 50))

      # First turn: push speech + silence
      Pipeline.push_frame(pid, speech_frame())
      Pipeline.push_frame(pid, silence_frame())

      # Immediately push speech again to interrupt before Task completes
      Process.sleep(10)
      Pipeline.push_frame(pid, speech_frame())
      Pipeline.push_frame(pid, silence_frame())

      # Allow both turns to resolve
      Process.sleep(400)

      metrics = Pipeline.get_metrics(pid)
      # At least one turn completed; errors should be 0 (interrupted turn is just cancelled)
      assert metrics.errors == 0
      assert metrics.audio_frames_processed >= 4

      Pipeline.stop(pid)
    end
  end

  describe "error handling" do
    test "STT error increments error metric and pipeline recovers" do
      config = %Pipeline.Config{
        stt: {FailingSTT, %{}},
        llm: {MockLLM, %{}},
        tts: {MockTTS, %{}},
        silence_ms: 50
      }

      {:ok, pid} = Pipeline.start_link(config)
      Pipeline.push_frame(pid, speech_frame())
      Pipeline.push_frame(pid, silence_frame())
      Process.sleep(300)

      metrics = Pipeline.get_metrics(pid)
      assert metrics.errors == 1
      assert metrics.turns_processed == 0
      assert Process.alive?(pid)

      Pipeline.stop(pid)
    end
  end

  describe "get_metrics/1" do
    test "returns metrics map with expected keys" do
      {:ok, pid} = Pipeline.start_link(base_config())
      metrics = Pipeline.get_metrics(pid)

      assert is_map(metrics)
      assert Map.has_key?(metrics, :turns_processed)
      assert Map.has_key?(metrics, :audio_frames_processed)
      assert Map.has_key?(metrics, :errors)

      Pipeline.stop(pid)
    end
  end
end
