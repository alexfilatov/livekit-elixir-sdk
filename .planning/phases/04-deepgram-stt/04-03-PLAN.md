defmodule Livekit.Agents.STT.AudioBufferTest do
  use ExUnit.Case, async: true

  alias Livekit.Agents.STT.AudioBuffer

  describe "new/1" do
    test "defaults min_duration_ms to 100" do
      buf = AudioBuffer.new()
      assert buf.min_duration_ms == 100
    end

    test "accepts custom min_duration_ms" do
      buf = AudioBuffer.new(min_duration_ms: 500)
      assert buf.min_duration_ms == 500
    end

    test "starts with empty data and zero duration" do
      buf = AudioBuffer.new()
      assert buf.data == <<>>
      assert buf.duration_ms == 0.0
    end
  end

  describe "push/3" do
    test "accumulates data binary" do
      buf = AudioBuffer.new()
      audio = <<1, 2, 3, 4>>
      buf2 = AudioBuffer.push(buf, audio, 48_000)
      assert buf2.data == audio
    end

    test "concatenates multiple pushes" do
      buf = AudioBuffer.new()
      buf2 = AudioBuffer.push(buf, <<1, 2>>, 48_000)
      buf3 = AudioBuffer.push(buf2, <<3, 4>>, 48_000)
      assert buf3.data == <<1, 2, 3, 4>>
    end

    test "increases duration_ms correctly for 16-bit mono PCM at 48kHz" do
      buf = AudioBuffer.new()
      # 9600 bytes = 4800 samples at 48kHz = 100ms
      audio = :binary.copy(<<0, 0>>, 4800)
      buf2 = AudioBuffer.push(buf, audio, 48_000)
      assert_in_delta buf2.duration_ms, 100.0, 0.1
    end

    test "accumulates duration_ms across multiple pushes" do
      buf = AudioBuffer.new()
      audio_50ms = :binary.copy(<<0, 0>>, 2400) # 50ms at 48kHz
      buf2 = AudioBuffer.push(buf, audio_50ms, 48_000)
      buf3 = AudioBuffer.push(buf2, audio_50ms, 48_000)
      assert_in_delta buf3.duration_ms, 100.0, 0.1
    end
  end

  describe "flush_if_ready/1" do
    test "returns :buffering when under threshold" do
      buf = AudioBuffer.new(min_duration_ms: 200)
      audio = :binary.copy(<<0, 0>>, 2400) # 50ms
      buf2 = AudioBuffer.push(buf, audio, 48_000)
      assert {:buffering, ^buf2} = AudioBuffer.flush_if_ready(buf2)
    end

    test "returns :ready when threshold met" do
      buf = AudioBuffer.new(min_duration_ms: 100)
      audio = :binary.copy(<<0, 0>>, 4800) # 100ms
      buf2 = AudioBuffer.push(buf, audio, 48_000)
      assert {:ready, ^audio, reset_buf} = AudioBuffer.flush_if_ready(buf2)
      assert reset_buf.data == <<>>
      assert reset_buf.duration_ms == 0.0
    end

    test "reset buffer preserves min_duration_ms setting" do
      buf = AudioBuffer.new(min_duration_ms: 250)
      audio = :binary.copy(<<0, 0>>, 4800) # 100ms
      buf2 = AudioBuffer.push(buf, audio, 48_000)
      buf3 = AudioBuffer.push(buf2, audio, 48_000)
      buf4 = AudioBuffer.push(buf3, audio, 48_000)
      {:ready, _, reset_buf} = AudioBuffer.flush_if_ready(buf4)
      assert reset_buf.min_duration_ms == 250
    end
  end

  describe "flush/1" do
    test "always returns accumulated data" do
      buf = AudioBuffer.new(min_duration_ms: 1000)
      audio = <<1, 2, 3, 4>>
      buf2 = AudioBuffer.push(buf, audio, 48_000)
      assert {^audio, reset_buf} = AudioBuffer.flush(buf2)
      assert reset_buf.data == <<>>
      assert reset_buf.duration_ms == 0.0
    end

    test "flush on empty buffer returns empty binary" do
      buf = AudioBuffer.new()
      assert {<<>>, _reset} = AudioBuffer.flush(buf)
    end
  end
end

defmodule Livekit.Agents.STT.DeepgramStreamTest do
  use ExUnit.Case, async: true

  alias Livekit.Agents.STT.DeepgramStream
  alias Livekit.Agents.STT.Deepgram.Config
  alias Livekit.Agents.STT.SpeechEvent

  @mock_config %Config{mock: true, language: "en-US"}

  defp collect_events(timeout \\ 500) do
    collect_events([], timeout)
  end

  defp collect_events(acc, timeout) do
    receive do
      {:speech_event, %SpeechEvent{type: :end} = event} ->
        Enum.reverse([event | acc])

      {:speech_event, event} ->
        collect_events([event | acc], timeout)
    after
      timeout -> Enum.reverse(acc)
    end
  end

  describe "mock mode streaming" do
    test "emits start event" do
      {:ok, _pid} = DeepgramStream.start_link({@mock_config, self()})
      events = collect_events()
      types = Enum.map(events, & &1.type)
      assert :start in types
    end

    test "emits interim event with text 'Hello'" do
      {:ok, _pid} = DeepgramStream.start_link({@mock_config, self()})
      events = collect_events()
      interim = Enum.find(events, &(&1.type == :interim))
      assert interim != nil
      assert interim.text == "Hello"
    end

    test "emits final event with longer transcript" do
      {:ok, _pid} = DeepgramStream.start_link({@mock_config, self()})
      events = collect_events()
      final = Enum.find(events, &(&1.type == :final))
      assert final != nil
      assert String.length(final.text) > 0
    end

    test "emits :end as last event" do
      {:ok, _pid} = DeepgramStream.start_link({@mock_config, self()})
      events = collect_events()
      assert List.last(events).type == :end
    end

    test "events arrive in order: start -> interim -> final -> end" do
      {:ok, _pid} = DeepgramStream.start_link({@mock_config, self()})
      events = collect_events()
      types = Enum.map(events, & &1.type)
      assert types == [:start, :interim, :final, :end]
    end

    test "event language matches config" do
      config = %Config{mock: true, language: "fr"}
      {:ok, _pid} = DeepgramStream.start_link({config, self()})
      events = collect_events()
      non_end = Enum.reject(events, &(&1.type == :end))
      Enum.each(non_end, fn event ->
        if event.text != "" do
          assert event.language == "fr"
        end
      end)
    end

    test "send_audio/2 does not crash in mock mode" do
      # In mock mode the GenServer exits via {:stop, :normal} immediately
      # send_audio is a no-op cast; test that calling it raises no errors
      # by checking the mock stream still delivers events
      {:ok, pid} = DeepgramStream.start_link({@mock_config, self()})
      # pid may already be dead in mock mode (depends on implementation)
      # cast is safe even to a dead pid via GenServer.cast
      DeepgramStream.send_audio(pid, <<0, 1, 2, 3>>)
      events = collect_events()
      assert Enum.any?(events, &(&1.type == :end))
    end
  end

  describe "Deepgram.stream/1 integration (mock)" do
    test "stream/1 returns {:ok, pid}" do
      alias Livekit.Agents.STT.Deepgram
      assert {:ok, _pid} = Deepgram.stream(@mock_config)
      # Drain mailbox
      collect_events()
    end

    test "stream/1 delivers full event sequence to caller" do
      alias Livekit.Agents.STT.Deepgram
      {:ok, _pid} = Deepgram.stream(@mock_config)
      events = collect_events()
      types = Enum.map(events, & &1.type)
      assert :start in types
      assert :final in types
      assert :end in types
    end
  end
end
