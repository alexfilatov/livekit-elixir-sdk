Create `lib/livekit/agents/tts/openai/cache.ex`. This is a standalone Agent-based
cache. It is NOT a globally supervised process — callers start it themselves with
`Cache.start_link/1` and pass the pid to subsequent calls.

Module: `Livekit.Agents.TTS.OpenAI.Cache`

**State shape** (Agent state is a map):
```elixir
%{
  entries: %{key => {value, inserted_at_monotonic_ms}},
  insertion_order: [key],   # oldest-first list for eviction
  ttl_ms: integer(),        # from opts[:ttl_seconds] * 1000, default 3600_000
  max_entries: integer()    # from opts[:max_entries], default 500
}
```

**Public API** (all functions take `cache_pid` as first arg):

`start_link(opts)` — starts the Agent:
```elixir
@spec start_link(keyword()) :: {:ok, pid()} | {:error, term()}
def start_link(opts \\ []) do
  ttl_ms = Keyword.get(opts, :ttl_seconds, 3600) * 1000
  max_entries = Keyword.get(opts, :max_entries, 500)
  Agent.start_link(fn -> %{entries: %{}, insertion_order: [], ttl_ms: ttl_ms, max_entries: max_entries} end)
end
```

`get(cache_pid, key)` — returns `{:ok, value}` if present and not expired, else `:miss`:
```elixir
@spec get(pid(), String.t()) :: {:ok, binary()} | :miss
def get(cache_pid, key) do
  Agent.get(cache_pid, fn state ->
    now = System.monotonic_time(:millisecond)
    case Map.get(state.entries, key) do
      nil -> :miss
      {value, inserted_at} ->
        if now - inserted_at <= state.ttl_ms, do: {:ok, value}, else: :miss
    end
  end)
end
```

`put(cache_pid, key, value)` — stores value; evicts oldest entry if at max_entries:
```elixir
@spec put(pid(), String.t(), binary()) :: :ok
def put(cache_pid, key, value) do
  Agent.update(cache_pid, fn state ->
    now = System.monotonic_time(:millisecond)
    # Remove existing entry from insertion_order if key already present
    new_order = List.delete(state.insertion_order, key)
    # Check if we need to evict oldest
    {evict_key, new_order2} =
      if length(new_order) >= state.max_entries do
        [oldest | rest] = new_order
        {oldest, rest}
      else
        {nil, new_order}
      end
    new_entries = state.entries
      |> Map.delete(evict_key)   # no-op if evict_key is nil
      |> Map.put(key, {value, now})
    %{state | entries: new_entries, insertion_order: new_order2 ++ [key]}
  end)
end
```

`clear(cache_pid)` — resets to empty:
```elixir
@spec clear(pid()) :: :ok
def clear(cache_pid) do
  Agent.update(cache_pid, fn state -> %{state | entries: %{}, insertion_order: []} end)
end
```

**synthesize/2:**
```elixir
@doc """
Synthesizes `text` into audio binary.

Options:
  - `:config` (required) — `%OpenAI.Config{}` with provider settings
  - `:voice` — override config voice (atom, e.g. `:alloy`)
  - `:format` — override config response_format (atom, e.g. `:mp3`)
  - `:cache` — pid of a `Cache` process; if provided, caching is active
"""
@impl true
@spec synthesize(String.t(), keyword()) :: {:ok, binary()} | {:error, term()}
def synthesize(text, opts \\ []) do
  config = Keyword.fetch!(opts, :config)
  voice = Keyword.get(opts, :voice, config.voice)
  format = Keyword.get(opts, :format, config.response_format)
  cache_pid = Keyword.get(opts, :cache)
  effective_config = %{config | voice: voice, response_format: format}

  if mock_mode?(effective_config) do
    {:ok, mock_synthesize(text, effective_config)}
  else
    cache_key = build_cache_key(effective_config, text)
    case maybe_cache_get(cache_pid, cache_key) do
      {:ok, cached} ->
        Logger.debug("TTS cache hit for text: #{String.slice(text, 0, 50)}")
        {:ok, cached}
      :miss ->
        case do_synthesize(effective_config, text) do
          {:ok, audio} ->
            maybe_cache_put(cache_pid, cache_key, audio)
            {:ok, audio}
          {:error, _} = err -> err
        end
    end
  end
end
```

`do_synthesize/2` — real HTTP call:
```elixir
defp do_synthesize(%Config{} = config, text) do
  client = build_client(config)
  body = %{
    model: Atom.to_string(config.model) |> String.replace("_", "-"),
    input: text,
    voice: Atom.to_string(config.voice),
    response_format: Atom.to_string(config.response_format),
    speed: config.speed
  }
  case Tesla.post(client, "/v1/audio/speech", body) do
    {:ok, %{status: 200, body: audio_binary}} -> {:ok, audio_binary}
    {:ok, %{status: status, body: body}} ->
      Logger.error("OpenAI TTS API error #{status}: #{inspect(body)}")
      {:error, {:api_error, status, body}}
    {:error, reason} ->
      Logger.error("OpenAI TTS request failed: #{inspect(reason)}")
      {:error, reason}
  end
end
```

IMPORTANT: The `/v1/audio/speech` endpoint returns raw audio bytes, NOT JSON.
Do NOT include `Tesla.Middleware.JSON` in `build_client/1`. The response body will be
the raw binary from Tesla without JSON decoding.

`build_client/1`:
```elixir
defp build_client(%Config{api_key: key, base_url: base_url}) do
  middleware = [
    {Tesla.Middleware.BaseUrl, base_url},
    {Tesla.Middleware.Headers, [
      {"Authorization", "Bearer #{key}"},
      {"Content-Type", "application/json"}
    ]},
    Tesla.Middleware.JSON   # encodes the REQUEST body only; response body stays binary
  ]
  Tesla.client(middleware, Tesla.Adapter.Hackney)
end
```

NOTE: `Tesla.Middleware.JSON` will attempt to decode the response body as JSON and
fail for binary audio. Use `Tesla.Middleware.JSON` for encoding the request body and
handle the raw binary response directly. To prevent response decoding, pass opts to
the middleware:
```elixir
{Tesla.Middleware.JSON, decode_content_types: ["application/json"]}
```
This ensures only `application/json` responses get JSON-decoded; audio bytes pass through raw.

defp build_cache_key(config, text) do

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| caller -> synthesize/2 | text and opts arrive from application code; text is sent verbatim to OpenAI |
| Tesla -> OpenAI API | API key in Authorization header; response body is raw audio binary |
| Cache Agent -> callers | Cache stores audio binaries in-memory; no persistence, no cross-process boundary |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-01 | Information Disclosure | build_client/1 | mitigate | API key is in Bearer header only; Tesla.Middleware.Logger is NOT included. Logger.error calls log status/reason but never the Authorization header. |
| T-06-02 | Spoofing | do_synthesize/2 | accept | Response comes from OpenAI over TLS; Hackney validates server cert by default. |
| T-06-03 | Denial of Service | Cache Agent | accept | Cache is bounded by max_entries (default 500) with oldest-entry eviction. Max memory bounded by max_entries * avg_audio_size. |
| T-06-04 | Information Disclosure | build_cache_key/2 | accept | Cache key is SHA256 of (model+voice+speed+format+text); original text not recoverable from key. |
| T-06-05 | Denial of Service | do_synthesize/2 | mitigate | Hackney default recv_timeout caps hanging requests. Large text inputs are accepted by OpenAI up to 4096 chars — no client-side truncation needed per API spec. |
</threat_model>
